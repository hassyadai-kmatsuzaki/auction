```typescript
// features/live-control/hooks/useLiveControl.ts
export function useLiveControl(auctionId: number) {
  const query = useQuery({
    queryKey: ['admin-live-state', auctionId],
    queryFn: () => adminAuctionApi.getLiveState(auctionId),
    refetchInterval: 3000,
    staleTime: 1000,
  });

  const pauseMutation = useMutation({
    mutationFn: () => adminAuctionApi.pause(auctionId),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['admin-live-state', auctionId] }),
  });

  const resumeMutation = useMutation({
    mutationFn: () => adminAuctionApi.resume(auctionId),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['admin-live-state', auctionId] }),
  });

  return {
    liveState: query.data,
    isLoading: query.isLoading,
    pause: pauseMutation.mutate,
    resume: resumeMutation.mutate,
    isPausing: pauseMutation.isPending,
    isResuming: resumeMutation.isPending,
  };
}
```

```typescript
// features/auction-form/hooks/useAuctionForm.ts
export function useAuctionForm(auctionId?: number) {
  const queryClient = useQueryClient();
  const isEdit = Boolean(auctionId);

  // 既存データ取得（編集時）
  const { data: initialData } = useQuery({
    queryKey: ['admin-auction', auctionId],
    queryFn: () => adminAuctionApi.getOne(auctionId!),
    enabled: isEdit,
  });

  // フォーム状態はローカルuseStateで管理
  const [formData, setFormData] = useState<AuctionFormData>(defaultFormData);

  // 保存Mutation
  const saveMutation = useMutation({
    mutationFn: (payload: AuctionFormData) =>
      isEdit ? adminAuctionApi.update(auctionId!, payload) : adminAuctionApi.create(payload),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['admin-auctions'] });
    },
  });

  return { formData, setFormData, save: saveMutation.mutate, isSaving: saveMutation.isPending };
}
```

## 7. Phase 4: パフォーマンス最適化

### 7-1. React.memoの適用基準

以下の条件を満たすコンポーネントに `React.memo` を適用する:

```
✅ React.memo 適用すべき:
- props が頻繁に変わらない純粋なコンポーネント
- リストの各要素（カード1枚、行1行）
- カウントダウン表示（CountdownChip ✅ 適用済み）
- 状態が変化しないUIパーツ（ボタン、バッジ等）

❌ React.memo 不要:
- 毎回 propsが変わるコンポーネント（効果なし）
- 軽量コンポーネント（コスト対効果が低い）
- ref を使うコンポーネント
```

### 7-2. TanStack Queryの `staleTime` 設定基準

```typescript
// データの変化頻度に応じて設定
staleTime: 500,        // ライブ状態（頻繁に変化）
staleTime: 5_000,      // 落札一覧（たまに変化）
staleTime: 60_000,     // オークション一覧（あまり変化しない）
staleTime: 5 * 60_000, // システム設定（ほぼ変化しない）
staleTime: Infinity,   // 静的データ（変化しない）
```

### 7-3. 重いコンポーネントの遅延読み込み


## 参考: 実装済みのベストプラクティス例

### 楽観的更新（useBidToggle.ts）

```typescript
// APIレスポンスを待たずにUIを先行更新 → 失敗時は自動ロールバック
onMutate: async ({ itemId, isActive }) => {
  await queryClient.cancelQueries({ queryKey: LIVE_STATE_QUERY_KEY(auctionId) });
  const previousData = queryClient.getQueryData<LiveState>(LIVE_STATE_QUERY_KEY(auctionId));

  // 楽観的更新
  queryClient.setQueryData<LiveState>(LIVE_STATE_QUERY_KEY(auctionId), (prev) => ({
    ...prev!,
    lanes: prev!.lanes.map((lane) =>
      lane.current_item?.id === itemId
        ? { ...lane, current_item: { ...lane.current_item, my_bid_status: isActive ? 'active' : 'inactive' } }
        : lane
    ),
  }));

  return { previousData };
},
onError: (_err, _vars, context) => {
  // 失敗時にロールバック
  if (context?.previousData) {
    queryClient.setQueryData(LIVE_STATE_QUERY_KEY(auctionId), context.previousData);
  }
},
onSettled: () => {
  // 成功・失敗どちらでもサーバーから最新データを再取得
  queryClient.invalidateQueries({ queryKey: LIVE_STATE_QUERY_KEY(auctionId) });
},
```

```typescript
// WebSocketイベントでキャッシュを直接更新 → APIリクエスト発生しない
const applyCountdownTick = (event: CountdownTickEvent) => {
  queryClient.setQueryData<LiveState>(LIVE_STATE_QUERY_KEY(auctionId), (prev) => {
    if (!prev) return prev;
    return {
      ...prev,
      lanes: prev.lanes.map((lane) =>
        lane.lane_id === event.lane_id && lane.current_item
          ? { ...lane, current_item: { ...lane.current_item, countdown_seconds: event.remaining_seconds } }
          : lane
      ),
    };
  });
};
```
